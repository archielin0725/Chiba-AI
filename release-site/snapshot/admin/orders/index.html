☁️ Google 試算表自動同步設定 (Option 2 Connector)

透過 Google Apps Script Webhook，前台顧客下單與後台訂單異動均可自動非同步推送到您的私密 Google 試算表，並可同時觸發 Gmail 或 LINE 即時通知。

設定後將儲存於您的本機安全環境中。每當有新訂單成立，系統將自動 POST 送出。

📋 Google Apps Script 後端程式碼範本 (Code.gs)：

在 Google 試算表中點選「擴充功能 ➔ Apps Script」，將下列程式碼貼上並點擊「部署 ➔ 新增部署作業 ➔ 網頁應用程式 (誰可以存取：所有人)」即可取得 Webhook 網址：

function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (sheet.getLastRow() === 0) { sheet.appendRow([ "時間", "訂單編號", "顧客姓名", "手機", "Email", "配送超商", "門市地址", "商品清單", "小計", "促銷折扣", "折扣碼", "實付總額", "狀態", "發票載具/統編" ]); } var data = JSON.parse(e.postData.contents); var itemsStr = (data.items || []).map(function(i) { return i.name + " (" + (i.size || "") + ") x" + i.quantity; }).join("; "); sheet.appendRow([ new Date().toLocaleString("zh-TW", {timeZone: "Asia/Taipei"}), data.orderId, data.customerName, data.phone, data.email, data.storeName || data.shippingMethod, data.storeAddr || "", itemsStr, data.subtotal, data.discount, data.promoCode || "", data.total, data.status || "預購成立", data.invoiceCarrier || data.invoiceType || "" ]); return ContentService.createTextOutput(JSON.stringify({result: "success"})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({result: "error", error: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
